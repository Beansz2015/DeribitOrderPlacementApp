' ClientWebSocket stand-in: records every frame the app sends; can be told to fail (the real
' SendWebSocketMessageAsync at frmMainPageV2.vb:1293-1314 then takes its own Catch arm).
Public Class FakeSocket
    Public State As WebSocketState = WebSocketState.Open
    Public FailSends As Boolean = False
    Public ReadOnly Sent As New List(Of (T As DateTime, Json As String))
    Public Function SendAsync(buffer As ArraySegment(Of Byte), type As WebSocketMessageType, endOfMessage As Boolean, token As CancellationToken) As Task
        If FailSends OrElse State <> WebSocketState.Open Then
            State = WebSocketState.Aborted
            Throw New WebSocketException("simulated: connection reset while sending")
        End If
        SyncLock Sent
            Sent.Add((DateTime.UtcNow, Encoding.UTF8.GetString(buffer.Array, buffer.Offset, buffer.Count)))
        End SyncLock
        Return Task.CompletedTask
    End Function
End Class
